const id = z.string().min(1).max(256);
export const borrowerActionSchema = z.discriminatedUnion('kind', [
  z.strictObject({kind:z.literal('reply'),message:z.string().min(1).max(6000)}),
  z.strictObject({kind:z.literal('investigate'),goal:z.string().min(1).max(2000),requirementIds:z.array(id).max(100),documentIds:z.array(id).max(100),message:z.string().min(1).max(2000)}),
  z.strictObject({kind:z.literal('answer_question'),questionId:id,sourceMessageId:id,answerText:z.string().min(1).max(4000),message:z.string().min(1).max(2000)}),
]);
export type BorrowerAction = z.infer<typeof borrowerActionSchema>;

/** Every native tool calls the authorized durable controller, not a second model.
 * One native tool per generation keeps spend and effect commits inside our ledger.
 * The outer durable loop supplies observations to the next generation. */
export const AGENT_LIMITS = Object.freeze({model:PILOT_MODEL.nativeId,maxOutputTokens:PILOT_MODEL.outputTokens,modelTimeoutMs:60000,maxInvestigationCalls:24,maxBorrowerCalls:1,maxRunMs:300000,maximumCallMicros:PILOT_MODEL.reservationMicros});
export interface AgentBindings<A> {model?:AgentConfig['model'];execute:(action:A)=>Promise<unknown>}
function nativeTools<A>(schemas:z.ZodObject<any>[],parse:(value:unknown)=>A,names:Record<string,string>,descriptions:Record<string,string>,execute:(action:A)=>Promise<unknown>):ToolsInput {
 let called=false;
 return Object.fromEntries(schemas.map(schema=>{
  const kind=schema.shape.kind.value as string,name=names[kind]??kind;
  return [name,createTool({id:name,description:descriptions[kind]??name,inputSchema:schema.omit({kind:true}),execute:async input=>{
   if(called)throw Error('Only one tool action per durable step');called=true;
   return execute(parse({...input,kind}));
  }})];
 }));
}
export function createBorrowerPreparationAgent(bindings:AgentBindings<BorrowerAction>) {
 const tools=nativeTools(borrowerActionSchema.options,borrowerActionSchema.parse,{reply:'reply_to_borrower',investigate:'investigate_documents',answer_question:'answer_investigation_question'}, {
  reply:'Publish a concise grounded answer or focused question to the borrower.',
  investigate:'Delegate a goal and focus scope to the durable document investigation agent. Returns an investigation receipt; does not claim completion.',
  answer_question:'Forward the exact current borrower message to an outstanding question, preserving its attribution. Do not invent or paraphrase an answer.'
 },bindings.execute);
 const agent=new Agent({id:'borrower-preparation',name:'Borrower Preparation Agent',model:bindings.model??AGENT_LIMITS.model,instructions:borrowerInstructions,tools});
 agent.__setLogger(noopLogger);return agent;
}
export function createDocumentInvestigationAgent(bindings:AgentBindings<PlannerAction>) {
 const tools=nativeTools(plannerActionSchema.options,plannerActionSchema.parse,{read_pages:'read_evidence',ask:'request_clarification',assess:'publish_assessment',propose_mapping:'propose_document_mapping'}, {
  inspect_document:'Inspect structure and number of addressable pages or record chunks. This does not establish coverage.',
  read_pages:'Read selected PDF pages or non-PDF record chunks. Choose native text, PDF/image rendering, or selective PDF/image OCR. Returned locations identify paragraphs, sheet cells and email lines.',
  search_evidence:'Search native text in selected pages or record chunks. Hits are locators, not inspected evidence; read the matching units before citing.',
  propose_mapping:'Record evidence-backed relevance to a lender request. Server policy owns application of the mapping.',
  ask:'Persist a focused question for the borrower preparation agent. Work can resume after the attributed answer.',
  assess:'Publish changed criterion findings and optionally reaffirm prior criterion IDs after reviewing changed context. Omit unchanged reusable findings. Backend preserves safe findings, marks unchecked criteria unknown and aggregates completeness; this does not require a full assessment.'
 },bindings.execute);
 const agent=new Agent({id:'document-investigation',name:'Document Investigation Agent',model:bindings.model??AGENT_LIMITS.model,instructions:investigationInstructions,tools});
 agent.__setLogger(noopLogger);return agent;
}
